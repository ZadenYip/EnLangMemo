Table words {
  // 官方词典整数主键
  word_id integer [primary key]
  spelling text [not null]
  entry_version integer [not null, note: '词条内容版本；官方词典内容发生变化时递增']
  phonetic_bre text
  phonetic_ame text

  // Epoch MS
  created_at integer [not null]
  updated_at integer [not null]

  indexes {
    spelling [unique, name: "idx_words_spelling"]
  }

  note: '单词表 (Word)'
}

// 2. 词性表 (Part of Speech)
Table word_poses {
  // 官方词典整数主键
  pose_id integer [primary key]
  word_id integer [not null]
  part_of_speech text

  created_at integer [not null]
  updated_at integer [not null]
  indexes {
    word_id [name: "idx_word_poses_word_id"]
  }
}
Ref: words.word_id < word_poses.word_id [delete: cascade]

// 3. 释义表
Table definitions {
  // 官方词典整数主键
  def_id integer [primary key] 
  word_pos_id integer [not null, note: '关联到具体的"词性"']
  def_src text [note: '源语言释义']
  def_tgt text [note: '目标语言释义']

  // Epoch MS
  created_at integer [not null]
  updated_at integer [not null]
  indexes {
    word_pos_id [name: "idx_definitions_word_pos_id"]
  }
}
Ref: word_poses.pose_id < definitions.word_pos_id [delete: cascade]


// 4. 例句表 (Example)
Table examples {
  exp_id integer [primary key, note: '官方词典整数主键']
  def_id integer [not null]
  ex_src text [not null, note: '源语言例句']
  ex_tgt text [note: '目标语言例句']

  indexes {
    def_id [name: "idx_examples_def_id"]
  }

  // Epoch MS
  created_at integer [not null]
  updated_at integer [not null]
}
Ref: definitions.def_id < examples.def_id [delete: cascade]

  sqlite_schema_version integer [not null, note: '本地应用 sqlite schema 版本']

  config jsonb [not null, note: 
  '''

export interface CollectionConfig {
    /**
    * IANA 
    */
    timeZone: string;
    /**
     * 复习时间重置 1 表示 1:00（24小时制）重置

}

Table decks {

  usn integer [not null]
  name text [not null]
  updated_at integer [not null]
  new_cards_per_day integer [not null, default: 20, note: "一天能学多少张新卡"]
  new_learned_today integer [not null, default: 0, note: "今天新卡学了多少个"]
  learned_today integer [not null, default: 0, note: "今天学习了多少次卡片"]
  reviewed_today integer [not null, default: 0, note: "今天复习了多少张卡片"]
  config jsonb [not null, note:

  export interface DeckConfig {
    /**
     * The maximum number of new cards can be learned per day in this deck.
     * -1 means no limit

    // TODO FSRS algorithm parameters
  }
  ''']
}

Table note_types {
  id blob [primary key, note: '笔记模板 id UUIDv7']
  name text [not null, note: '模板名']
  preset_template_id integer [not null, default: 0, note: '官方发放的预设模板编号；0 表示未分配/未知预设，1 表示 Sentence Mining Definition']
  usn integer [not null, note: 
  '''
    用来对比
      -- update sequence number : used to figure out diffs when syncing. 
      --   value of -1 indicates changes that need to be pushed to server. 
      --   usn < server usn indicates changes that need to be pulled from server.
  ''']
  updated_at integer [not null]

  note_template jsonb [
    not null,
    note: '''
    {
      "css": "CSS 模板"
      // 单个展示模板，直接展开在笔记模板上
      "front": "{{TargetWord}}", 
      "back": "xxx",
      // 放在第一个的字段
      "sortField": timestamp,
      // 有哪些字段
      "fields": [{"id": timestamp, "name": "TargetWord"}]
    }
    '''
  ]
  Note: '笔记模板；一类笔记只对应一种展示模板'
}

Table notes {
  id blob [primary key, note: '笔记 id UUIDv7']
  note_type_id blob [not null]
  usn integer [not null]
  created_at integer [not null]
  updated_at integer [not null]
  sense_id integer [note: '释义追踪 ID，对应官方词典 definitions.def_id，用于研究记忆迁移率']
  sort_field text [note: '用来排序的字段，也得是清洗后的']
  search_fields text [note: 'fields 清洗后的所有值拼凑']
  fields jsonb [not null, note: 
  '''
  Key-Value
  {id} 是对应 note type 里 fields 存的 id
  实际填入的内容，例如 {"{id}": "Apple"}
  ''']

  indexes {
    (usn) [name: 'ix_notes_usn']
  }

  Note: '笔记'
}
Ref: notes.note_type_id > note_types.id
